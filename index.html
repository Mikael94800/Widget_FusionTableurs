async function executeImport() { const selects = document.querySelectorAll('.mapping-select'); const mapping = {}; selects.forEach(select => { const originalHeader = select.closest('.mapping-card').dataset.original; const targetCol = select.value; if (targetCol) { mapping[originalHeader] = targetCol; } }); if (Object.keys(mapping).length === 0) { alert("Veuillez associer au moins une colonne avant d'importer."); return; } const btn = document.getElementById('btnImport'); btn.disabled = true; document.getElementById('status-msg').innerText = "Préparation des données..."; // Liste des colonnes de type "Référence multiple" nécessitant le format Grist ['L', id1, id2, ...] const multiRefColumns = [ "Listes_Diffusion_Intervenants", "Listes_Diffusion_Cinema", "Listes_Diffusion_Direction", "Listes_Diffusion_Houdremont" ]; const recordsToInsert = parsedData.map(row => { const newRecord = {}; for (const [origKey, targetKey] of Object.entries(mapping)) { let val = row[origKey]; if (val !== undefined && val !== "") { // Gestion spécifique pour formater correctement les listes de diffusion (Références multiples) if (multiRefColumns.includes(targetKey)) { if (typeof val === 'string') { // Sépare par des virgules ou points-virgules et convertit en nombres entiers const ids = val.split(/[,;]/).map(v => parseInt(v.trim(), 10)).filter(n => !isNaN(n)); newRecord[targetKey] = ['L', ...ids]; } else if (Array.isArray(val)) { const ids = val.map(v => parseInt(v, 10)).filter(n => !isNaN(n)); newRecord[targetKey] = ['L', ...ids]; } else { newRecord[targetKey] = val; } } else { newRecord[targetKey] = val; } } } return newRecord; }); // Filtrer pour ne garder que les objets non vides const validRecords = recordsToInsert.filter(rec => Object.keys(rec).length > 0); console.log(`Total de lignes valides à importer : ${validRecords.length}`); console.log("Exemple du premier enregistrement propre :", validRecords[0]); try { const batchSize = 100; for (let i = 0; i < validRecords.length; i += batchSize) { const batch = validRecords.slice(i, i + batchSize); document.getElementById('status-msg').innerText = `Envoi en cours (${Math.min(i + batchSize, validRecords.length)} / ${validRecords.length})...`; console.log(`Envoi du lot ${i} à ${i + batch.length}...`); await grist.selectedTable.create(batch); } document.getElementById('status-msg').innerText = `Succès ! ${validRecords.length} contacts ont été importés.`; document.getElementById('status-msg').style.color = "var(--success)"; setTimeout(() => { location.reload(); }, 2500); } catch (err) { console.error("Erreur critique attrapée :", err); alert("Échec de l'importation : " + (err.message || JSON.stringify(err))); btn.disabled = false; document.getElementById('status-msg').innerText = "Échec de l'importation."; } }
